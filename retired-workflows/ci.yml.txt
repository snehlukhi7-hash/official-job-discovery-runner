name: Public discovery CI

on:
  pull_request:
  push:
    branches: [main]

permissions:
  contents: read

jobs:
  verify:
    runs-on: ubuntu-latest
    timeout-minutes: 10
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"
          cache: pip
      - run: python -m pip install -e '.[test]'
      - run: python -m pytest -q
      - run: python scripts/privacy_scan.py .
      - run: python -m compileall -q discovery_runner
